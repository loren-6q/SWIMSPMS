const { onRequest, onCall, HttpsError } = require("firebase-functions/v2/https"); const { onTaskDispatched } = require("firebase-functions/v2/tasks"); const { defineSecret } = require("firebase-functions/params"); const { getFunctions } = require("firebase-admin/functions"); const admin = require("firebase-admin"); const axios = require("axios"); const { GoogleGenerativeAI, HarmCategory, HarmBlockThreshold } = require("@google/generative-ai"); if (!admin.apps.length) { admin.initializeApp(); } const db = admin.firestore(); const telegramToken = defineSecret("TELEGRAM_BOT_TOKEN"); const geminiApiKey = defineSecret("GEMINI_API_KEY"); const stripeSecret = defineSecret("STRIPE_SECRET_KEY"); function sanitizeSecret(secret) { return secret ? secret.trim() : ""; } function sanitizeName(name) { if (!name) return ""; return name .normalize("NFD") .replace(/[\u0300-\u036f]/g, "") .replace(/[^a-zA-Z\s]/g, "") .replace(/\s+/g, " ") .trim() .toUpperCase(); } async function sendTelegramText(chatId, text, token) { try { await axios.post(`https://api.telegram.org/bot${token}/sendMessage`, { chat_id: chatId, text: text, parse_mode: "Markdown" }); } catch (e) { console.error("Failed to send Telegram message:", e.message); } } // 0. LIVE STRIPE FUNCTION (Untouched) exports.createSetupIntent = onCall({ secrets: [stripeSecret] }, async (request) => { try { const stripe = require("stripe")(sanitizeSecret(stripeSecret.value())); const setupIntent = await stripe.setupIntents.create({ payment_method_types: ['card'], }); return { clientSecret: setupIntent.client_secret }; } catch (error) { console.error("Stripe Error:", error); throw new HttpsError('internal', error.message); } }); // 1. LIGHTNING FAST WEBHOOK exports.telegramBotWebhook = onRequest({ secrets: [telegramToken, geminiApiKey], region: "asia-southeast1", memory: "256MiB", timeoutSeconds: 30 }, async (req, res) => { try { const payload = req.body; if (!payload || !payload.message) return res.status(200).send({ ok: true }); const message = payload.message; const chatId = message.chat.id; const messageId = message.message_id.toString(); const botToken = sanitizeSecret(telegramToken.value()); // Deduplication const dedupeRef = db.collection("system").doc("telegram_cache").collection("processed").doc(messageId); const dedupeSnap = await dedupeRef.get(); if (dedupeSnap.exists) return res.status(200).send({ ok: true }); await dedupeRef.set({ processedAt: admin.firestore.FieldValue.serverTimestamp() }); // Handle Property Settings if (message.text && message.text.startsWith("/setproperty")) { const propertyName = message.text.split(" ")[1]; if (!propertyName) { await sendTelegramText(chatId, "⚠️ Please specify a property.\n\nExample: `/setproperty SWIMS`", botToken); return res.status(200).send({ ok: true }); } await db.collection("system").doc("chat_mappings").collection("chats").doc(chatId.toString()).set({ property: propertyName.toUpperCase() }); await sendTelegramText(chatId, `✅ Success! This group chat is now linked to: *${propertyName.toUpperCase()}*`, botToken); return res.status(200).send({ ok: true }); } // Handle Passports if (message.photo) { let captionText = message.caption || ""; let mediaGroupId = message.media_group_id || null; let roomNumber = ""; if (mediaGroupId) { const mgRef = db.collection("system").doc("media_groups").collection("groups").doc(mediaGroupId); if (captionText) { roomNumber = captionText.trim(); await mgRef.set({ roomNumber: roomNumber, timestamp: admin.firestore.FieldValue.serverTimestamp() }); } else { await new Promise(r => setTimeout(r, 1000)); const mgSnap = await mgRef.get(); if (mgSnap.exists) roomNumber = mgSnap.data().roomNumber; } } else if (captionText) { roomNumber = captionText.trim(); } const roomDisplay = roomNumber ? `\n🏷️ *Room Tag:* ${roomNumber}` : ""; // Instantly acknowledge to Telegram await sendTelegramText(chatId, `📥 Passport queued for processing...${roomDisplay}`, botToken); // Push to V2 Task Queue safely (Targeting exact region) const queue = getFunctions().taskQueue("locations/asia-southeast1/functions/processPassportTask"); await queue.enqueue({ message: message, chatId: chatId, roomNumber: roomNumber }); return res.status(200).send({ ok: true }); } return res.status(200).send({ ok: true }); } catch (error) { console.error("Webhook Global Error:", error); return res.status(200).send({ ok: true }); } }); // 2. V2 TASK QUEUE (Max 2 processing at a time = NO RATE LIMIT CRASHES) exports.processPassportTask = onTaskDispatched({ secrets: [telegramToken, geminiApiKey], region: "asia-southeast1", memory: "1GiB", timeoutSeconds: 120, rateLimits: { maxConcurrentDispatches: 2 // Guarantee we NEVER hit the 15/min API limit } }, async (req) => { const { message, chatId, roomNumber } = req.data; const botToken = sanitizeSecret(telegramToken.value()); const geminiKey = sanitizeSecret(geminiApiKey.value()); try { const photoArray = message.photo; const targetPhoto = photoArray[photoArray.length - 1]; const infoResponse = await axios.get(`https://api.telegram.org/bot${botToken}/getFile?file_id=${targetPhoto.file_id}`); const filePath = infoResponse.data.result.file_path; const fileUrl = `https://api.telegram.org/file/bot${botToken}/${filePath}`; const downloadResponse = await axios.get(fileUrl, { responseType: "arraybuffer" }); const imageBase64 = Buffer.from(downloadResponse.data).toString("base64"); const genAI = new GoogleGenerativeAI(geminiKey); const model = genAI.getGenerativeModel({ model: "gemini-3.6-flash", generationConfig: { responseMimeType: "application/json" }, safetySettings: [ { category: HarmCategory.HARM_CATEGORY_HARASSMENT, threshold: HarmBlockThreshold.BLOCK_NONE }, { category: HarmCategory.HARM_CATEGORY_HATE_SPEECH, threshold: HarmBlockThreshold.BLOCK_NONE }, { category: HarmCategory.HARM_CATEGORY_SEXUALLY_EXPLICIT, threshold: HarmBlockThreshold.BLOCK_NONE }, { category: HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT, threshold: HarmBlockThreshold.BLOCK_NONE } ] }); const prompt = `Analyze this passport biographical page. Extract the fields and return them strictly inside a valid flat JSON block containing: firstName (First name only), middleName (Any remaining words in the "Given Names" field. If none, leave blank), lastName (The "Surname" field), gender (Look at the "Sex" field. Output exactly "M" or "F"), passportNumber (The document number), nationality (CRITICAL: Output ONLY the 3-letter ISO 3166-1 alpha-3 country code. Example: USA, SWE, THA, GBR), dobDay (Strictly 2-digit day of birth), dobMonth (Strictly 2-digit month of birth), dobYear (4-digit year of birth). Do not include Markdown formatting. Output only RAW JSON.`; const result = await model.generateContent([ prompt, { inlineData: { data: imageBase64, mimeType: "image/jpeg" } } ]); let outputText = result.response.text(); outputText = outputText.replace(/`{3}[a-zA-Z]*\n?/gi, '').replace(/`{3}\n?/g, '').trim(); const guestData = JSON.parse(outputText); const appId = 'hotel-pms-v1'; const cleanPassport = (guestData.passportNumber || "").toUpperCase().replace(/\s/g, ''); if (cleanPassport && cleanPassport.length > 3) { const submissionsRef = db.collection("artifacts").doc(appId).collection("public").doc("data").collection("submissions"); const dupQuery = await submissionsRef.where("passportNumber", "==", cleanPassport).get(); let isDuplicate = false; dupQuery.forEach(doc => { if (doc.data().status !== 'completed' && doc.data().status !== 'failed') isDuplicate = true; }); if (isDuplicate) { await sendTelegramText(chatId, `⚠️ *Duplicate Blocked:* ${guestData.firstName} is already waiting in the Dashboard!`, botToken); return; } } let chatProperty = "UNASSIGNED"; const mappingSnap = await db.collection("system").doc("chat_mappings").collection("chats").doc(chatId.toString()).get(); if (mappingSnap.exists) chatProperty = mappingSnap.data().property; // Use raw 3-letter code for Nationality and single character for Gender const rawNat = (guestData.nationality || "").toUpperCase().trim(); const docRef = await db.collection("artifacts").doc(appId).collection("public").doc("data").collection("submissions").add({ property: chatProperty, roomNumber: roomNumber, firstName: sanitizeName(guestData.firstName), middleName: sanitizeName(guestData.middleName), lastName: sanitizeName(guestData.lastName), passportNumber: cleanPassport, nationality: rawNat, dobDay: guestData.dobDay || "", dobMonth: guestData.dobMonth || "", dobYear: guestData.dobYear || "", gender: (guestData.gender || "M").toUpperCase().charAt(0), checkoutDate: "", phoneNo: "", status: "processing", chatId: chatId, createdAt: admin.firestore.FieldValue.serverTimestamp() }); const roomDisplayFinal = roomNumber ? `\n• *Room Tag:* ${roomNumber}` : ""; const summaryText = `✅ *Passport Processed!*\n` + `• *Name:* ${guestData.firstName || ""} ${guestData.lastName || ""}\n` + `• *Nat:* ${rawNat}\n` + `• *Passport No:* \`${cleanPassport || "N/A"}\`${roomDisplayFinal}\n` + `⚙ *Pushed to Web Dashboard.*`; await sendTelegramText(chatId, summaryText, botToken); try { const scraperModule = require("./scraper"); await scraperModule.executeTM30Submission(appId, docRef.id); } catch (err) { console.error("Scraper logic error:", err); } } catch (err) { console.error("Worker Execution Error:", err); await sendTelegramText(chatId, `❌ *Critical Error:* ${err.message}`, botToken); } });
